cmake_minimum_required(VERSION 3.24)
project(DiskPilot VERSION 0.1.0 LANGUAGES CXX)

option(DISKPILOT_BUILD_TESTS "Build GoogleTest tests (downloads pinned source)" ON)
option(DISKPILOT_BUILD_GUI "Build the Qt 6 desktop application" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Threads REQUIRED)
include(FetchContent)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nlohmann_json
  GIT_REPOSITORY https://github.com/nlohmann/json.git
  GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03
  GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(nlohmann_json)

add_library(diskpilot_core src/core/version.cpp src/core/scanner.cpp src/core/analysis.cpp src/core/treemap.cpp
    src/core/file_access.cpp src/core/duplicates.cpp src/core/cleanup.cpp src/core/reports.cpp)
target_link_libraries(diskpilot_core PRIVATE nlohmann_json::nlohmann_json)
target_link_libraries(diskpilot_core PUBLIC Threads::Threads)
target_include_directories(diskpilot_core PUBLIC include)
target_compile_features(diskpilot_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(diskpilot_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(diskpilot_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(DISKPILOT_BUILD_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
  qt_standard_project_setup(REQUIRES 6.5)
  qt_add_executable(DiskPilot src/gui/main.cpp src/gui/controller.cpp src/gui/controller.hpp)
  qt_add_qml_module(DiskPilot URI DiskPilotUI VERSION 1.0 QML_FILES src/gui/Main.qml)
  target_link_libraries(DiskPilot PRIVATE diskpilot_core Qt6::Quick Qt6::QuickControls2)
  if(WIN32)
    set_target_properties(DiskPilot PROPERTIES WIN32_EXECUTABLE TRUE)
    if(MSVC)
      target_sources(DiskPilot PRIVATE src/cli/diskpilot.manifest)
    endif()
  endif()
  install(TARGETS DiskPilot RUNTIME DESTINATION bin)
  qt_generate_deploy_qml_app_script(TARGET DiskPilot OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()

add_executable(diskpilot-cli src/cli/main.cpp)
if(WIN32)
  target_compile_definitions(diskpilot_core PRIVATE _WIN32_WINNT=0x0602)
  if(MINGW)
    target_link_options(diskpilot-cli PRIVATE -municode)
  endif()
endif()
target_link_libraries(diskpilot-cli PRIVATE diskpilot_core)
if(MSVC)
  target_sources(diskpilot-cli PRIVATE src/cli/diskpilot.manifest)
endif()

if(DISKPILOT_BUILD_TESTS)
  include(CTest)
  include(FetchContent)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571
    GIT_SHALLOW FALSE)
  FetchContent_MakeAvailable(googletest)
  add_executable(diskpilot_tests tests/version_test.cpp tests/scanner_test.cpp tests/analysis_test.cpp tests/treemap_test.cpp
      tests/duplicates_test.cpp tests/cleanup_test.cpp tests/reports_test.cpp)
  target_link_libraries(diskpilot_tests PRIVATE diskpilot_core GTest::gtest_main nlohmann_json::nlohmann_json)
  include(GoogleTest)
  gtest_discover_tests(diskpilot_tests)
  if(DISKPILOT_BUILD_GUI)
    add_test(NAME gui_load COMMAND DiskPilot --smoke-test)
    set_tests_properties(gui_load PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 30)
  endif()
  add_test(NAME cli_help COMMAND diskpilot-cli --help)
  set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "Usage:")
  add_test(NAME cli_version COMMAND diskpilot-cli --version)
  set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "0.1.0")
  add_test(NAME cli_unknown_command COMMAND diskpilot-cli invalid-command)
  set_tests_properties(cli_unknown_command PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli_scan_exports COMMAND ${CMAKE_COMMAND}
    -DCLI=$<TARGET_FILE:diskpilot-cli>
    -DFIXTURE_PARENT=${CMAKE_CURRENT_BINARY_DIR}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.cmake)
endif()

install(TARGETS diskpilot-cli RUNTIME DESTINATION bin)
install(FILES LICENSE README.md SECURITY.md docs/DISTRIBUTION.md DESTINATION share/doc/DiskPilot)
set(CPACK_PACKAGE_NAME DiskPilot)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_GENERATOR ZIP)
include(CPack)
